<ExamQuestion title="Question 1: Measuring Efficiency [10 Marks]">
  <SubQuestion part="a" prompt={<>Define time complexity and explain what input size <code>N</code> represents. [4]</>} solution={<>Time complexity describes how an algorithm's running time grows as the input grows. N is a measure of the amount of input, such as the number of elements in an array.</>} />
  <SubQuestion part="b" prompt={<>Give two limitations of comparing algorithms only by timing Java programs. [4]</>} solution={<>Any two: the algorithms must first be implemented; results depend on hardware or software; programming style can affect time; and the tested datasets may not represent every possible input.</>} />
  <SubQuestion part="c" prompt={<>What is the main benefit of the methodological approach? [2]</>} solution={<>It estimates scalability by counting operations, so it can compare algorithms independently of a particular machine and before full implementation.</>} />
</ExamQuestion>

<ExamQuestion title="Question 2: Best, Average, and Worst Cases [9 Marks]">
  <SubQuestion part="a" prompt={<>Define the best case, average case, and worst case for an input of size N. [6]</>} solution={<>Best case is the shortest possible runtime, average case is the expected runtime across possible inputs, and worst case is the maximum runtime.</>} />
  <SubQuestion part="b" prompt={<>Which case is most commonly used in this course, and why is it useful? [3]</>} solution={<>The worst case. It gives an upper limit on the work and is useful when we must plan for the slowest valid input.</>} />
</ExamQuestion>

<ExamQuestion title="Question 3: Reading Code Structure [12 Marks]">
  <SubQuestion part="a" prompt={<>A method contains four simple sequential statements and no loops. What is its Big-O complexity? Explain. [3]</>} solution={<>O(1). The number of operations is fixed and does not grow with N.</>} />
  <SubQuestion part="b" prompt={<>A loop processes every element of an array once. What is its Big-O complexity? [3]</>} solution={<>O(N), because the loop body executes once per element.</>} />
  <SubQuestion part="c" prompt={<>Two nested loops each run N times. How many times does the inner statement execute, and what is the Big-O complexity? [3]</>} solution={<>The inner statement executes N x N = N^2 times, so the complexity is O(N^2).</>} />
  <SubQuestion part="d" prompt={<>For an <code>if</code>/<code>else</code>, which branch is counted in a worst-case estimate? [3]</>} solution={<>Count the condition and the branch with the greater runtime.</>} />
</ExamQuestion>

<ExamQuestion title="Question 4: Simplifying Growth Functions [12 Marks]">
  <SubQuestion part="a" prompt={<>Simplify <code>7N - 2</code> to Big-O notation. [3]</>} solution={<>O(N). Drop the constant coefficient and constant term.</>} />
  <SubQuestion part="b" prompt={<>Simplify <code>3N^3 + 20N^2 + 5</code>. [3]</>} solution={<>O(N^3). The cubic term grows fastest.</>} />
  <SubQuestion part="c" prompt={<>Simplify <code>3 log N + 5</code>. [3]</>} solution={<>O(log N). Drop the coefficient and constant term.</>} />
  <SubQuestion part="d" prompt={<>Simplify <code>8N^2 log N + 5N^2 + N</code>. [3]</>} solution={<>O(N^2 log N). It is the dominant term.</>} />
</ExamQuestion>

<ExamQuestion title="Question 5: Applying Big-O [9 Marks]">
  <SubQuestion part="a" prompt={<>State the time complexity of an algorithm that computes the average of N numbers using one loop. [3]</>} solution={<>O(N), because every number must be read and added once.</>} />
  <SubQuestion part="b" prompt={<>State the time complexity of printing every character in a string of length N. [3]</>} solution={<>O(N), because each of the N characters is processed once.</>} />
  <SubQuestion part="c" prompt={<>State the time complexity of computing N factorial iteratively with one loop. [3]</>} solution={<>O(N), because the loop performs one multiplication for each value from 1 through N.</>} />
</ExamQuestion>
